from __future__ import annotations

import pytest

    AuthorizationDenied,

class ReadStore:
    """Authority fake with an explicit commit boundary and recorded decisions."""

    def __init__(self, build, policy):  # type: ignore[no-untyped-def]
        self.owner = build.scope()
        self.user = build.principal()
        self.release_id = policy.release.release_id
        self.authority = PrincipalAuthority(self.user, 4)
        self.slices = (build.grant(self.owner, "TenantMember"),)
        self.slot = build.envelope(self.owner, ResourceType.FACT_SLOT)
        self.scope = build.envelope(self.owner, classifications=("restricted",))
        self.resources = {self.slot.key: self.slot, self.scope.key: self.scope}
        self.pending = []
        self.committed = []
        self.commits = 0
        self.fail_commit = False
        self.hydrated = []
        self.revocation = None

    def current_policy_release(self):  # type: ignore[no-untyped-def]
        return self.release_id

    def peek_delegation(self, delegation_id):  # type: ignore[no-untyped-def]
        return None

    def lock_authority(self, principal_ids):  # type: ignore[no-untyped-def]
        return {self.user.id: self.authority}

    def lock_delegation(self, delegation_id):  # type: ignore[no-untyped-def]
        return None

    def scope_slices(self, principal_id):  # type: ignore[no-untyped-def]
        return self.slices

    def hydrate(self, key, *, require_record=True):  # type: ignore[no-untyped-def]
        self.hydrated.append(key)
        return self.resources.get(key)

    async def run_sync(self, callback):  # type: ignore[no-untyped-def]
        return callback(self)

    async def append_decision(self, decision):  # type: ignore[no-untyped-def]
        self.pending.append(decision)

    async def commit(self):  # type: ignore[no-untyped-def]
        if self.fail_commit:
            raise OSError("synthetic audit failure")
        self.committed.extend(self.pending)
        self.pending = []
        self.commits += 1
        if self.revocation is not None:
            self.revocation()

    async def rollback(self):  # type: ignore[no-untyped-def]
        self.pending = []


async def test_exact_payload_dependencies_share_one_successful_commit(build, policy):
    store = ReadStore(build, policy)
    execution = AuthorizationExecution(store, policy, policy.release, clock=lambda: build.now)
    selected = {"scope": store.scope.key, "pages": [7, 8]}
    loads = []

    async def load(decision):
        assert decision.resource == store.slot.key and store.commits == 0
        loads.append(True)
        return selected

    async def scope_reads(payload):
        assert payload is selected and store.commits == 0
        return [payload["scope"], payload["scope"]]

    def revoke_after_locks_release():
        store.authority = PrincipalAuthority(store.user, 5)
        store.slices = (build.grant(store.owner, "TenantMember", classifications=("internal",)),)

    store.revocation = revoke_after_locks_release
    result = await execution.read(
        store.user,
        store.slot.key,
        Action.READ,
        trace_id="group",
        load=load,
        additional_reads=scope_reads,
    )
    assert result is selected and loads == [True] and store.commits == 1
    assert {d.resource for d in store.committed} == {store.slot.key, store.scope.key}
    assert {d.grant_revision for d in store.committed} == {4}
    assert all(d.operation_outcome == OperationOutcome.SUCCEEDED for d in store.committed)
    # Revocation after the completed transaction affects the next operation.
    with pytest.raises(AuthorizationDenied):
        await execution.read(store.user, store.scope.key, Action.READ, trace_id="next", load=load)


async def test_denied_scope_returns_no_payload_and_audits_both_decisions(build, policy):
    store = ReadStore(build, policy)
    store.slices = (build.grant(store.owner, "TenantMember", classifications=("internal",)),)
    execution = AuthorizationExecution(store, policy, policy.release, clock=lambda: build.now)

    async def load(_decision):
        return {"scope": store.scope.key}

    async def scopes(payload):
        return [payload["scope"]]

    with pytest.raises(AuthorizationDenied) as denied:
        await execution.read(
            store.user,
            store.slot.key,
            Action.READ,
            trace_id="denied",
            load=load,
            additional_reads=scopes,
        )
    assert denied.value.decision.resource == store.scope.key
    assert store.commits == 1
    assert {d.resource: d.operation_outcome for d in store.committed} == {
        store.slot.key: OperationOutcome.FAILED,
        store.scope.key: OperationOutcome.DENIED,
    }


async def test_denied_primary_does_not_load_or_discover_scope(build, policy):
    store = ReadStore(build, policy)
    store.slices = (build.grant(store.owner, "Reviewer"),)
    execution = AuthorizationExecution(store, policy, policy.release, clock=lambda: build.now)

    async def forbidden(_payload):
        pytest.fail("denied primary must not load a payload or discover its scope")

    with pytest.raises(AuthorizationDenied):
        await execution.read(
            store.user,
            store.slot.key,
            Action.READ,
            trace_id="primary",
            load=forbidden,
            additional_reads=forbidden,
        )
    assert store.scope.key not in store.hydrated
    assert len(store.committed) == 1


async def test_scope_discovery_failure_audits_failed_primary(build, policy):
    store = ReadStore(build, policy)
    execution = AuthorizationExecution(store, policy, policy.release, clock=lambda: build.now)

    async def load(_decision):
        return {"scope": store.scope.key}

    async def unavailable(_payload):
        raise LookupError("scope ownership metadata unavailable")

    with pytest.raises(LookupError):
        await execution.read(
            store.user,
            store.slot.key,
            Action.READ,
            trace_id="failed",
            load=load,
            additional_reads=unavailable,
        )
    assert [d.operation_outcome for d in store.committed] == [OperationOutcome.FAILED]


async def test_expiry_during_private_load_is_rechecked_before_release(build, policy):
    store = ReadStore(build, policy)
    store.slices = (
        build.grant(store.owner, "TenantMember", expires_at=build.now + timedelta(seconds=1)),
    )
    instant = [build.now]
    execution = AuthorizationExecution(store, policy, policy.release, clock=lambda: instant[0])

    async def load(_decision):
        return {"scope": store.scope.key}

    async def scopes(payload):
        instant[0] = build.now + timedelta(seconds=2)
        return [payload["scope"]]

    with pytest.raises(AuthorizationDenied):
        await execution.read(
            store.user,
            store.slot.key,
            Action.READ,
            trace_id="expired",
            load=load,
            additional_reads=scopes,
        )
    assert not any(d.operation_outcome == OperationOutcome.SUCCEEDED for d in store.committed)


async def test_group_audit_failure_never_returns_the_private_payload(build, policy):
    store = ReadStore(build, policy)
    store.fail_commit = True
    execution = AuthorizationExecution(store, policy, policy.release, clock=lambda: build.now)

    async def load(_decision):
        return {"scope": store.scope.key}

    async def scopes(payload):
        return [payload["scope"]]

    with pytest.raises(AuthorizationUnavailable):
        await execution.read(
            store.user,
            store.slot.key,
            Action.READ,
            trace_id="audit",
            load=load,
            additional_reads=scopes,
        )
    assert store.committed == [] and store.pending == [] and store.commits == 0

async def _scoped_answer(w):  # type: ignore[no-untyped-def]
    slot = (await w.resolve(w.key(w.a, AGGREGATE_LIMIT))).slot_id
    secret = w.artifact(w.a, classifications=("restricted",))
    envelope = {
        "answer_contract": "fact-answer/v1",
        "state": "NOT_FOUND",
        "scope": {
            "content_artifact_id": str(secret),
            "profile_release_ref": "profile/1",
            "pages": [7, 8],
        },
    }
    response = await w.commit(slot, envelope)
    assert response.status_code == 201, response.text
    return slot, secret, envelope


def _reader(w, *, broad: bool):  # type: ignore[no-untyped-def]
    with Session(w.sync_engine) as session, session.begin():
        reader = fixtures.seed_principal(
            session, issuer="urn:f0007:read", subject=uuid4().hex, kind=PrincipalKind.USER
        )
        fixtures.seed_grant(
            session, reader.id, *w.a.scope, "TenantMember", classifications=("internal",)
        )
        broad_grant = (
            fixtures.seed_grant(session, reader.id, *w.a.scope, "TenantMember") if broad else None
        )
    return reader, broad_grant


async def test_http_authorizes_the_scope_of_the_exact_selected_payload(slot_world, monkeypatch):
    w = slot_world
    slot, secret, envelope = await _scoped_answer(w)
    reader, _ = _reader(w, broad=False)
    # Simulate a pre-query seeing an earlier, readable scope while the selected
    # HTTP version contains the restricted scope. No pre-query is authoritative.
    earlier = {
        **envelope,
        "scope": {**envelope["scope"], "content_artifact_id": str(w.a.artifact_id)},
    }

    async def earlier_answer(self, slot_id, *, valid_at, known_at):
        return earlier

    monkeypatch.setattr(SqlAlchemyFactSlotStore, "answer_at", earlier_answer)
    response = await w.get(slot, principal=reader)
    assert response.status_code == 404
    assert response.json()["code"] == "not_found"
    assert str(secret) not in response.text
    assert (
        w.scalar(
            "SELECT count(*) FROM audit_event WHERE actor_principal_id = %s "
            "AND resource_id = %s AND action = 'read' AND operation_outcome = 'denied'",
            reader.id,
            secret,
        )
        == 1
    )


async def test_revocation_waits_for_slot_scope_and_payload_read_to_commit(slot_world):
    w = slot_world
    slot, secret, envelope = await _scoped_answer(w)
    reader, broad_grant = _reader(w, broad=True)
    revision = w.scalar(
        "SELECT revision FROM principal_authority WHERE principal_id = %s", reader.id
    )
    selected = asyncio.Event()
    attempted = asyncio.Event()
    selections = []

    class PausingAnswerStore(SqlAlchemyFactSlotStore):
        async def answer_at(self, slot_id, *, valid_at, known_at):
            stored = await super().answer_at(slot_id, valid_at=valid_at, known_at=known_at)
            selections.append(slot_id)
            if len(selections) == 1:
                selected.set()
                await attempted.wait()
            return stored

    def revoke(*, timeout: bool) -> bool:
        try:
            with Session(w.sync_engine) as session, session.begin():
                if timeout:
                    session.execute(text("SET LOCAL lock_timeout = '300ms'"))
                revoke_membership(
                    session,
                    broad_grant,
                    operator_id=fixtures.SYNTHETIC_OPERATOR,
                    approval_ref=fixtures.FIXTURE_APPROVAL,
                    at=datetime.now(UTC),
                )
        except OperationalError as exc:
            if timeout and getattr(exc.orig, "sqlstate", None) == "55P03":
                return True
            raise
        return False

    async def concurrent_revoke():
        await selected.wait()
        try:
            return await asyncio.to_thread(revoke, timeout=True)
        finally:
            attempted.set()

    async def read():
        now = datetime.now(UTC)
        async with AsyncSession(w.engine, expire_on_commit=False) as session:
            store = PausingAnswerStore(session)
            service = AuthorizedFactSlots(
                AuthorizationExecution(
                    SqlAlchemyAuthorityStore(session), w.policy, w.policy.release
                ),
                FactSlotResolver(w.registry, store),
                NameFactReader(store),
                FactAnswerReader(store),
            )
            return await service.read_answer(
                reader, slot, valid_at=now, known_at=now, trace_id="f0007-group-read"
            )

    answer, revocation_blocked = await asyncio.wait_for(
        asyncio.gather(read(), concurrent_revoke()), timeout=10
    )
    assert revocation_blocked is True
    assert selections == [slot]
    assert answer.to_json() == envelope
    decisions = w.rows(
        "SELECT resource_id, grant_revision, operation_outcome FROM audit_event "
        "WHERE trace_id = 'f0007-group-read' AND event_type = 'authorization_decision'"
    )
    assert {d[0] for d in decisions} == {slot, secret}
    assert {d[1] for d in decisions} == {revision}
    assert {d[2] for d in decisions} == {"succeeded"}
    # After the protected read releases its locks, supported revocation succeeds.
    assert await asyncio.to_thread(revoke, timeout=False) is False
    now = datetime.now(UTC)
    async with w.slots() as service:
        with pytest.raises(SlotNotFound):
            await service.read_answer(
                reader, slot, valid_at=now, known_at=now, trace_id="after-revocation"
            )
    assert (await w.get(slot, principal=reader)).status_code == 404
"""Require a complete legacy/v1 FactSlot identity shape (F0007 review FR-H2).

Revision 0012 is already shipped and stays unchanged. Its CHECK allowed SQL
UNKNOWN for incomplete v1 rows. This follow-up admits only TRUE, refuses any
existing malformed rows without repairing them, and preserves all data on both
upgrade and downgrade. No RLS, grants, ownership keys or version rows change.

Revision ID: 0013
Revises: 0012
"""

import logging
from collections.abc import Sequence

from alembic import op
from sqlalchemy import text

revision: str = "0013"
down_revision: str | None = "0012"
branch_labels: Sequence[str] | str | None = None
depends_on: Sequence[str] | str | None = None

log = logging.getLogger("alembic.runtime.migration")

# WHY: migration history is immutable. The old expression is frozen here so a
# downgrade restores exactly 0012's semantics without importing mutable code.
PREVIOUS_CHECK = r"""
  (identity_version IS NULL AND slot_role IS NULL AND predicate_key IS NULL
     AND predicate_release_ref IS NULL AND cardinality IS NULL AND object_entity_id IS NULL
     AND parent_slot_id IS NULL AND member_key IS NULL AND fingerprint_version IS NULL
     AND identity_fingerprint IS NULL AND created_by IS NULL)
  OR (identity_version = 1 AND fingerprint_version = 1 AND created_by IS NOT NULL
     AND identity_fingerprint ~ '^[0-9a-f]{64}$'
     AND predicate_key ~ '^[a-z][a-z0-9-]{0,31}:[A-Za-z][A-Za-z0-9]{0,63}@v[1-9][0-9]{0,3}$'
     AND length(predicate_release_ref) > 0 AND slot_type = predicate_key
     AND cardinality IN ('SINGLE','MULTI','RELATIONSHIP')
     AND ((slot_role = 'SLOT' AND parent_slot_id IS NULL AND member_key IS NULL
            AND ((cardinality = 'RELATIONSHIP') = (object_entity_id IS NOT NULL)))
       OR (slot_role = 'MEMBER' AND cardinality = 'MULTI' AND parent_slot_id IS NOT NULL
            AND length(member_key) > 0 AND object_entity_id IS NULL)))
"""

IDENTITY_SHAPE = r"""
  (identity_version IS NULL AND slot_role IS NULL AND predicate_key IS NULL
     AND predicate_release_ref IS NULL AND cardinality IS NULL AND object_entity_id IS NULL
     AND parent_slot_id IS NULL AND member_key IS NULL AND fingerprint_version IS NULL
     AND identity_fingerprint IS NULL AND created_by IS NULL)
  OR (identity_version IS NOT NULL AND identity_version = 1
     AND fingerprint_version IS NOT NULL AND fingerprint_version = 1
     AND created_by IS NOT NULL AND slot_role IS NOT NULL
     AND identity_fingerprint IS NOT NULL AND identity_fingerprint ~ '^[0-9a-f]{64}$'
     AND predicate_key IS NOT NULL
     AND predicate_key ~ '^[a-z][a-z0-9-]{0,31}:[A-Za-z][A-Za-z0-9]{0,63}@v[1-9][0-9]{0,3}$'
     AND predicate_release_ref IS NOT NULL AND length(predicate_release_ref) > 0
     AND slot_type = predicate_key AND cardinality IS NOT NULL
     AND cardinality IN ('SINGLE','MULTI','RELATIONSHIP')
     AND ((slot_role = 'SLOT' AND parent_slot_id IS NULL AND member_key IS NULL
            AND ((cardinality = 'RELATIONSHIP') = (object_entity_id IS NOT NULL)))
       OR (slot_role = 'MEMBER' AND cardinality = 'MULTI' AND parent_slot_id IS NOT NULL
            AND member_key IS NOT NULL AND length(member_key) > 0 AND object_entity_id IS NULL)))
"""


def upgrade() -> None:
    bind = op.get_bind()
    # WHY: hold the same lock through validation and replacement so another
    # writer cannot insert an invalid shape between the scan and the new CHECK.
    bind.execute(text("LOCK TABLE fact_slot IN ACCESS EXCLUSIVE MODE"))
    invalid = f"({IDENTITY_SHAPE}) IS NOT TRUE"
    malformed = bind.execute(text(f"SELECT count(*) FROM fact_slot WHERE {invalid}")).scalar_one()
    if malformed:
        identifiers = bind.execute(
            text(f"SELECT id FROM fact_slot WHERE {invalid} ORDER BY id LIMIT 10")
        ).scalars()
        samples = ", ".join(str(slot_id) for slot_id in identifiers)
        raise RuntimeError(
            f"0013 upgrade refused: {malformed} malformed fact_slot identity rows; "
            f"sample slot IDs: {samples}. No rows modified; explicit governed repair required."
        )
    op.execute("ALTER TABLE fact_slot DROP CONSTRAINT ck_fact_slot_identity")
    op.execute(
        f"ALTER TABLE fact_slot ADD CONSTRAINT ck_fact_slot_identity "
        f"CHECK (({IDENTITY_SHAPE}) IS TRUE)"
    )
    log.info("0013 identity guard installed; existing FactSlot rows preserved")


def downgrade() -> None:
    op.execute("ALTER TABLE fact_slot DROP CONSTRAINT ck_fact_slot_identity")
    op.execute(
        f"ALTER TABLE fact_slot ADD CONSTRAINT ck_fact_slot_identity CHECK ({PREVIOUS_CHECK})"
    )
