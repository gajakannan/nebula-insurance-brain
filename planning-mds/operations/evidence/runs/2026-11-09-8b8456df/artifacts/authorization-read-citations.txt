   325	    async def _safe_rollback(self) -> None:
   326	        # The original failure is what matters; a failed rollback adds nothing.
   327	        with contextlib.suppress(Exception):
   328	            await self._store.rollback()

   330	    async def _persist_fresh(self, decisions: Sequence[AuthorizationDecision]) -> None:
   331	        """Roll back any business work, then durably audit in a new transaction."""
   332	        await self._safe_rollback()
   333	        try:
   334	            for decision in decisions:
   335	                await self._store.append_decision(decision)
   336	            await self._store.commit()
   337	        except Exception as exc:  # noqa: BLE001
   338	            await self._safe_rollback()
   339	            raise AuthorizationUnavailable("decision audit could not be persisted") from exc

   341	    async def _deny(self, evaluated: Sequence[Evaluated]) -> AuthorizationDenied:
   342	        decisions = [
   343	            e.decision
   344	            if not e.decision.allowed
   345	            else e.decision.with_outcome(OperationOutcome.FAILED)
   346	            for e in evaluated
   347	        ]
   348	        await self._persist_fresh(decisions)
   349	        denied = next(d for d in decisions if not d.allowed)
   350	        return AuthorizationDenied(denied)
   351	
   352	    def _recheck_all(self, evaluated: Sequence[Evaluated]) -> list[Evaluated]:
   353	        at = self._clock()
   354	        return [recheck(e, self._policy, at) for e in evaluated]

   356	    async def read(
   357	        self,
   358	        authenticated: Principal,
   359	        key: ResourceKey,
   360	        action: str,
   361	        *,
   362	        trace_id: str,
   363	        load: Callable[[AuthorizationDecision], Awaitable[T]],
   364	        requested: RequestedScope = NO_FILTER,
   365	        delegation_id: UUID | None = None,
   366	    ) -> T:
   367	        evaluated = await self._guarded_decide(
   368	            authenticated, [key], action, requested, trace_id, delegation_id
   369	        )
   370	        if not evaluated[0].decision.allowed:
   371	            raise await self._deny(evaluated)
   372	        evaluated = self._recheck_all(evaluated)
   373	        if not evaluated[0].decision.allowed:
   374	            raise await self._deny(evaluated)
   375	        decision = evaluated[0].decision
   376	        try:
   377	            payload = await load(decision)
   378	        except Exception:
   379	            await self._persist_fresh([decision.with_outcome(OperationOutcome.FAILED)])
   380	            raise
   381	        try:
   382	            await self._store.append_decision(decision.with_outcome(OperationOutcome.SUCCEEDED))
   383	            await self._store.commit()
   384	        except Exception as exc:  # noqa: BLE001 - no payload leaves without durable audit
   385	            await self._safe_rollback()
   386	            raise AuthorizationUnavailable("read audit could not be persisted") from exc
   387	        return payload
