   132	    except UnicodeEncodeError as exc:

   137	def canonical_decimal(raw: object) -> str:
   138	    """One canonical decimal text: no exponent, no trailing fractional zeros, `-0`
   139	    becomes `0`. `1E+6`, `1000000.00` and `1000000` all become `1000000`.
   140	
   141	    The input must be a string. Digits are handled exactly, never through a
   142	    float or a precision-limited Decimal context."""
   143	    if not isinstance(raw, str) or len(raw) > DECIMAL_MAX_INPUT or not _DECIMAL.match(raw):
   144	        raise ValueRejected(ValueReason.DECIMAL_INVALID)
   145	    try:
   146	        sign, digit_tuple, exponent = Decimal(raw).as_tuple()
   147	    except InvalidOperation as exc:  # pragma: no cover - the pattern admits only finite forms
   148	        raise ValueRejected(ValueReason.DECIMAL_INVALID) from exc
   149	    assert isinstance(exponent, int)
   150	    digits = list(digit_tuple)
   151	    while len(digits) > 1 and digits[0] == 0:
   152	        digits.pop(0)
   153	    if all(d == 0 for d in digits):
   154	        return "0"
   155	    while exponent < 0 and digits[-1] == 0:
   156	        digits.pop()
   157	        exponent += 1
   158	    text = "".join(str(d) for d in digits)
   159	    if exponent >= 0:
   160	        body = text + "0" * exponent
   161	    elif -exponent >= len(text):
   162	        body = "0." + "0" * (-exponent - len(text)) + text
   163	    else:
   164	        body = text[:exponent] + "." + text[exponent:]
   165	    if len(body) > DECIMAL_MAX_CANONICAL:
   166	        raise ValueRejected(ValueReason.DECIMAL_INVALID)
   167	    return ("-" if sign else "") + body
